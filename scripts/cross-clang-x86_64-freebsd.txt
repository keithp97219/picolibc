# FreeBSD amd64, using libos/freebsd. Run this inside FreeBSD with the
# base system clang, so the compiler and its support library are the
# ones FreeBSD ships; libos/freebsd carries the kernel interface it uses
[binaries]
c = ['clang', '--target=x86_64-unknown-freebsd15.1', '-march=core2', '-mfpmath=sse', '-msse2', '-fno-pic', '-fno-PIE', '-static', '-nostdlib']
cpp = ['clang++', '--target=x86_64-unknown-freebsd15.1', '-march=core2', '-mfpmath=sse', '-msse2', '-fno-pic', '-fno-PIE', '-static', '-nostdlib']
c_ld = 'lld'
cpp_ld = 'lld'
ar = 'ar'
nm = 'nm'
strip = 'strip'
objcopy = 'objcopy'

[host_machine]
system='freebsd'
cpu_family='x86_64'
cpu='x86_64'
endian='little'

[built-in options]
# The compiler support library from the FreeBSD base system
c_link_args = ['-L/usr/lib']
cpp_link_args = ['-L/usr/lib']

[properties]
librt = '-lcompiler_rt'
skip_sanity_check = true
needs_exe_wrapper = false
link_spec = '--build-id=none'
crt0_default = 'crt0-freebsd'
oslib_default = 'freebsd'
linkerscript_default_c = 'picolibc_linux.ld'
linkerscript_default_cpp = 'picolibcpp_linux.ld'
target_c_args = ['-static', '-fno-pic', '-fno-PIE', '-nostdlib', '-nostartfiles', '-Wl,--build-id=none']
default_flash_addr = '0x00200000'
default_flash_size = '0x3fe00000'
default_ram_addr   = '0x40000000'
default_ram_size   = '0x3fff00000'
